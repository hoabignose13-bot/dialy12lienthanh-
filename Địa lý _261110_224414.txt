    function setupImageDrop(zoneId, inputId, previewId) {
      const zone = document.getElementById(zoneId);
      const input = document.getElementById(inputId);
      const preview = document.getElementById(previewId);
      if(!zone || !input) return;
      zone.onclick = function(e) { if(e.target !== input) input.click(); };
      zone.ondragover = function(e) { e.preventDefault(); zone.classList.add('dragover'); };
      zone.ondragleave = function() { zone.classList.remove('dragover'); };
      zone.ondrop = function(e) {
        e.preventDefault(); zone.classList.remove('dragover');
        if (e.dataTransfer.files.length > 0) {
          input.files = e.dataTransfer.files;
          const reader = new FileReader();
          reader.onload = function(ev) { preview.src = ev.target.result; };
          reader.readAsDataURL(input.files[0]);
        }
      };
      input.onchange = function() {
        if (input.files.length > 0) {
          const reader = new FileReader();
          reader.onload = function(ev) { preview.src = ev.target.result; };
          reader.readAsDataURL(input.files[0]);
        }
      };
    }

    async function uploadCloud(file, type, statusEl) {
      const fd = new FormData();
      fd.append("upload_preset", UPLOAD_PRESET);
      fd.append("file", file);
      return new Promise(function(resolve, reject) {
        const xhr = new XMLHttpRequest();
        xhr.open("POST", "https://api.cloudinary.com/v1_1/" + CLOUD_NAME + "/" + type + "/upload");
        xhr.upload.onprogress = function(e) {
          if (e.lengthComputable && statusEl) statusEl.textContent = "Đang tải lên: " + Math.round((e.loaded / e.total) * 100) + "%";
        };
        xhr.onload = function() {
          if (xhr.status === 200) resolve(JSON.parse(xhr.responseText).secure_url);
          else reject("Lỗi upload");
        };
        xhr.send(fd);
      });
    }

          if (indexB === -1) indexB = 99;

              <div class="book-card">

    function parseExamText(text) {
      const questions = [];
      const lines = text.split('\n').map(l => l.trim()).filter(l => l);
      
      let currentQ = null;
      let currentOptions = [];
      let currentCorrect = null;
      
      const reQuestion = /^(?:Câu|Cau)\s*\d+\s*[:.)]?\s*(.+)$/i;
      const reSimpleQ = /^(\d+)\s*[.):]\s+(.+)$/;
      const reOption = /^([A-D])\s*[.):]\s*(.+)$/i;
      const reAnswer = /^(?:Đáp án|Dap an|ĐA|DA|Answer|Correct)\s*[:.]?\s*([A-D])/i;
      
      for(const line of lines) {
        const qMatch = line.match(reQuestion);
        const sMatch = !qMatch && line.match(reSimpleQ);
        const aMatch = line.match(reAnswer);
        
        if(qMatch || sMatch) {
          if(currentQ && currentOptions.length >= 2 && currentCorrect !== null) {
            questions.push({ q: currentQ, options: currentOptions, correct: currentCorrect });
          }
          currentQ = qMatch ? qMatch[1].trim() : sMatch[2].trim();
          currentOptions = [];
          currentCorrect = null;
          continue;
        }
        
        if(aMatch && currentQ) {
          currentCorrect = aMatch[1].toUpperCase().charCodeAt(0) - 65;
          continue;
        }
        
        const oMatch = line.match(reOption);
        if(oMatch && currentQ) {
          let optText = oMatch[2].trim();
          if(/[✓✔✅]\s*$/.test(optText)) {
            optText = optText.replace(/[✓✔✅]\s*$/, '').trim();
            if(currentCorrect === null) currentCorrect = currentOptions.length;
          }
          currentOptions.push(optText);
          continue;
        }
        
        if(currentQ && currentOptions.length === 0) {
          currentQ += ' ' + line;
        }
      }
      
      if(currentQ && currentOptions.length >= 2 && currentCorrect !== null) {
        questions.push({ q: currentQ, options: currentOptions, correct: currentCorrect });
      }
      
      return questions;
    }

    function renderExams() {
      const container = document.getElementById('exam-list-grouped');
      if(!container) return;
      
      const filtered = currentExamFilter === 'All' ? globalExams : globalExams.filter(e => (e.category || 'Khác') === currentExamFilter);
      
      const cats = [...new Set(globalExams.map(e => e.category || 'Khác'))];
      const fb = document.getElementById('exam-filters');
      if(fb) {
        fb.innerHTML = `<div class="filter-btn ${currentExamFilter === 'All' ? 'active' : ''}" onclick="filterExams('All', this)">Tất cả (${globalExams.length})</div>` +
          cats.map(c => `<div class="filter-btn ${currentExamFilter === c ? 'active' : ''}" onclick="filterExams('${escapeHtml(c).replace(/'/g, "\\'")}', this)">${escapeHtml(c)}</div>`).join('');
      }
      
      if(filtered.length === 0) {
        container.innerHTML = "<p style='color:var(--text-muted); text-align:center; padding:40px;'>Chưa có đề thi nào trong mục này.</p>";
        return;
      }
      
      const grouped = {};
      filtered.forEach(e => {
        const cat = e.category || 'Khác';
        if(!grouped[cat]) grouped[cat] = [];
        grouped[cat].push(e);
      });
      
      let html = '';
      Object.keys(grouped).forEach(cat => {
        html += `<div class="group-header" style="color:#f39c12; border-color:rgba(243,156,18,0.3);"><i class="fa-solid fa-folder-open"></i> ${escapeHtml(cat)}</div>`;
        html += `<div class="book-grid" style="grid-template-columns: repeat(auto-fill, minmax(280px, 1fr)); margin-bottom:20px;">`;
        grouped[cat].forEach(exam => {
          const adminBtns = globalIsAdmin ? `
            <button class="btn btn-warning btn-sm" onclick="event.stopPropagation(); openExamEditor('${exam.id}')"><i class="fa-solid fa-pen"></i></button>
            <button class="btn btn-danger btn-sm" onclick="event.stopPropagation(); deleteExam('${exam.id}')"><i class="fa-solid fa-trash"></i></button>
          ` : '';
          const durationText = exam.duration > 0 ? `${exam.duration} phút` : 'Không giới hạn';
          html += `
            <div class="exam-card" onclick="openExamTaker('${exam.id}')">
              <div class="exam-icon"><i class="fa-solid fa-file-circle-check"></i></div>
              <h4>${escapeHtml(exam.title)}</h4>
              <div class="exam-meta">
                <span><i class="fa-solid fa-list-check"></i> ${exam.questions.length} câu</span>
                <span><i class="fa-solid fa-clock"></i> ${durationText}</span>
              </div>
              ${exam.description ? `<p style="color:var(--text-muted); font-size:13px; margin-bottom:10px;">${escapeHtml(exam.description)}</p>` : ''}
              <div class="exam-actions">
                <button class="btn btn-warning btn-sm" style="flex:1;"><i class="fa-solid fa-play"></i> Làm bài</button>
                ${adminBtns}
              </div>
            </div>
          `;
        });
        html += `</div>`;
      });
      container.innerHTML = html;
    }

    window.openExamTaker = function(examId) {
      const exam = globalExams.find(e => e.id === examId);
      if(!exam) return;
      currentExam = exam;
      window.examAnswers = {};
      
      document.getElementById('exam-taker-title').textContent = exam.title;
      document.getElementById('exam-taker-body').innerHTML = exam.questions.map((q, i) => `
        <div class="exam-question" data-qindex="${i}">
          <div class="exam-q-text"><strong>Câu ${i+1}:</strong> ${escapeHtml(q.q)}</div>
          <div class="exam-options">
            ${q.options.map((opt, j) => `
              <label class="exam-option" data-qindex="${i}" data-oindex="${j}">
                <input type="radio" name="q${i}" value="${j}" onchange="window.examAnswers[${i}]=${j}">
                <span>${String.fromCharCode(65+j)}. ${escapeHtml(opt)}</span>
              </label>
            `).join('')}
          </div>
        </div>
      `).join('');
      
      document.getElementById('exam-taker-result').style.display = 'none';
      document.getElementById('exam-taker-result').innerHTML = '';
      document.getElementById('exam-taker-body').style.display = 'block';
      document.getElementById('exam-taker-footer').style.display = 'flex';
      document.getElementById('exam-taker-modal').style.display = 'flex';
      
      if(examTimerInterval) clearInterval(examTimerInterval);
      const timerEl = document.getElementById('exam-taker-timer');
      if(exam.duration > 0) {
        timerEl.style.display = 'inline-block';
        let remaining = exam.duration * 60;
        const updateTimer = () => {
          const m = Math.floor(remaining / 60);
          const s = remaining % 60;
          timerEl.textContent = `⏱ ${m}:${String(s).padStart(2,'0')}`;
          if(remaining <= 0) {
            clearInterval(examTimerInterval);
            alert("⏰ Hết thời gian! Bài sẽ được nộp tự động.");
            window.submitExam(true);
            return;
          }
          remaining--;
        };
        updateTimer();
        examTimerInterval = setInterval(updateTimer, 1000);
      } else {
        timerEl.style.display = 'none';
      }
    };

    window.closeExamTaker = function() {
      document.getElementById('exam-taker-modal').style.display = 'none';
      if(examTimerInterval) { clearInterval(examTimerInterval); examTimerInterval = null; }
      currentExam = null;
      window.examAnswers = {};
    };

    window.submitExam = function(forceSubmit) {
      if(!currentExam) return;
      const total = currentExam.questions.length;
      const answered = Object.keys(window.examAnswers).length;
      
      if(!forceSubmit && answered < total && !confirm(`Bạn còn ${total - answered} câu chưa trả lời. Vẫn nộp?`)) return;
      
      if(examTimerInterval) { clearInterval(examTimerInterval); examTimerInterval = null; }
      document.getElementById('exam-taker-timer').style.display = 'none';
      
      let correct = 0;
      const results = currentExam.questions.map((q, i) => {
        const chosen = window.examAnswers[i];
        const isCorrect = chosen === q.correct;
        if(isCorrect) correct++;
        return { chosen, correct: q.correct, isCorrect };
      });
      
      const score = Math.round((correct / total) * 100) / 10;
      
      document.getElementById('exam-taker-body').innerHTML = currentExam.questions.map((q, i) => {
        const r = results[i];
        return `
          <div class="exam-question ${r.isCorrect ? 'correct' : 'wrong'}">
            <div class="exam-q-text"><strong>Câu ${i+1}:</strong> ${escapeHtml(q.q)} ${r.isCorrect ? '✅' : '❌'}</div>
            <div class="exam-options">
              ${q.options.map((opt, j) => {
                let cls = '';
                if(j === r.correct) cls = 'option-correct';
                else if(j === r.chosen) cls = 'option-wrong';
                return `<div class="exam-option ${cls}" style="cursor:default;"><span>${String.fromCharCode(65+j)}. ${escapeHtml(opt)} ${j === r.correct ? '✓' : ''}</span></div>`;
              }).join('')}
            </div>
            ${r.chosen !== undefined && !r.isCorrect ? `<div class="exam-note">Bạn chọn: ${String.fromCharCode(65+r.chosen)}</div>` : ''}
            ${r.chosen === undefined ? `<div class="exam-note">Bạn chưa trả lời câu này</div>` : ''}
          </div>
        `;
      }).join('');
      
      document.getElementById('exam-taker-result').innerHTML = `
        <div class="exam-summary">
          <h3>📊 Kết Quả Bài Làm</h3>
          <div class="exam-score">${score.toFixed(1)}<span style="font-size:24px; color:var(--text-muted);">/10</span></div>
          <p><b>Đúng: ${correct} / ${total} câu</b></p>
          <p>${score >= 8 ? '🎉 Xuất sắc! Em học rất tốt!' : score >= 5 ? '👍 Cố gắng thêm nhé!' : '💪 Em cần ôn lại bài!'}</p>
          <div style="display:flex; gap:10px; margin-top:20px; justify-content:center;">
            <button class="btn btn-outline" onclick="closeExamTaker()">Đóng</button>
            <button class="btn btn-warning" onclick="retakeExam()">🔄 Làm Lại</button>
          </div>
        </div>
      `;
      document.getElementById('exam-taker-result').style.display = 'block';
      document.getElementById('exam-taker-footer').style.display = 'none';
      
      const profile = JSON.parse(localStorage.getItem('h2_profile') || '{}');
      if(profile.name) {
        addDoc(collection(db, "exam_history"), {
          examId: currentExam.id,
          examTitle: currentExam.title,
          studentName: profile.name,
          studentClass: profile.class || "",
          studentSchool: profile.school || "",
          score, correct, total,
          submittedAt: serverTimestamp()
        }).catch(e => console.log("Lỗi lưu lịch sử:", e));
      }
    };

    window.retakeExam = function() {
      if(!currentExam) return;
      const id = currentExam.id;
      openExamTaker(id);
    };

    onSnapshot(query(collection(db, "exams"), orderBy("createdAt", "desc")), function(snap) {
      globalExams = snap.docs.map(d => ({ id: d.id, ...d.data() }));
      renderExams();
    }, function(err) {
      console.error("Lỗi load đề thi:", err);
    });

    setTimeout(() => {
      const ta = document.getElementById('exam-edit-questions');
      if(!ta) return;
      ta.addEventListener('dragover', e => { e.preventDefault(); ta.style.borderColor = '#f39c12'; });
      ta.addEventListener('dragleave', () => { ta.style.borderColor = 'var(--border)'; });
      ta.addEventListener('drop', e => {
        e.preventDefault();
        ta.style.borderColor = 'var(--border)';
        const file = e.dataTransfer.files[0];
        if(!file) return;
        if(!file.name.match(/\.(txt|text)$/i)) { alert("Chỉ hỗ trợ file .txt"); return; }
        const reader = new FileReader();
        reader.onload = ev => { ta.value = ev.target.result; window.previewExam(); };
        reader.readAsText(file, 'UTF-8');
      });
    }, 500);

  } catch (error) {
    console.error("Lỗi khởi tạo:", error);
    alert("Đã xảy ra lỗi hệ thống. Hãy nhấn F12 để xem chi tiết.");
  }
</script>
